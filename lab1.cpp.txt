#include <iostream>
#include <cmath>
using namespace std;

int main() {
    // Задача 1
    double a = 1, b = 0.4;
    double numerator = pow(a, 3) + 3 * pow(a, 2) * b;
    double denominator = 3 * a * pow(b, 2) + pow(b, 3);
    
    double y = pow(numerator, 1.0 / 3.0) / denominator;
    
    cout << "y = " << y << endl;

    // Задача 2
    double x, y1;
    cout << "Vvedite koordinaty tochki x i y: ";
    cin >> x >> y1;

    
    if (x <= 0 && y1 <= 0 && y1 >= -0.5 * x - 3) {
        cout << "TRUE" << endl;
    } else {
        cout << "FALSE" << endl;
    }

   
    int dummy;
    cout << "Press Enter to exit...";
    cin >> dummy;
    
    return 0;
}
